local function hashString(str)
	local hash = 5381
	for i = 1, #str do
		hash = (bit32 and bit32.bxor(hash * 33, string.byte(str, i))) or ((hash * 33) + string.byte(str, i))
	end
	return string.format("%X", math.abs(hash))
end

local function createKey(durationUnit, durationVal)
	local randomPart = string.format("%04X%04X", math.random(0, 65535), math.random(0, 65535))
	local expireTime
	if durationUnit == "Lifetime" then
		expireTime = "LIFETIME"
	else
		local seconds = 0
		if durationUnit == "Jours" then
			seconds = durationVal * 86400
		elseif durationUnit == "Mois" then
			seconds = durationVal * 30 * 86400
		elseif durationUnit == "Années" then
			seconds = durationVal * 365 * 86400
		end
		expireTime = tostring(os.time() + seconds)
	end
	local payload = "FK-" .. randomPart .. "-" .. expireTime
	local sig = hashString(payload .. SECRET_SALT)
	return payload .. "-" .. sig
end

local function verifyKey(keyStr)
	if not keyStr or type(keyStr) ~= "string" then return false, "Clé invalide" end
	keyStr = keyStr:gsub("%s+", "")

	if blockedKeys[keyStr] then
		return false, "Clé bloquée par l'Admin"
	end

	local parts = string.split(keyStr, "-")
	if #parts ~= 4 or parts[1] ~= "FK" then return false, "Format de clé incorrect" end

	local randomPart = parts[2]
	local expireStr = parts[3]
	local sig = parts[4]

	local expectedPayload = "FK-" .. randomPart .. "-" .. expireStr
	local expectedSig = hashString(expectedPayload .. SECRET_SALT)

	if sig ~= expectedSig then
		return false, "Clé invalide ou falsifiée"
	end

	if expireStr == "LIFETIME" then
		return true, "Lifetime (Infinie)", math.huge
	end

	local expireTime = tonumber(expireStr)
	if not expireTime then return false, "Clé corrompue" end

	local now = os.time()
	local remaining = expireTime - now
	if remaining <= 0 then
		return false, "Clé expirée", 0
	end

	local days = math.floor(remaining / 86400)
	local hours = math.floor((remaining % 86400) / 3600)
	local mins = math.floor((remaining % 3600) / 60)
	local secs = remaining % 60

	local timeFormatted = ""
	if days > 0 then timeFormatted = timeFormatted .. days .. "j " end
	if hours > 0 or days > 0 then timeFormatted = timeFormatted .. hours .. "h " end
	if mins > 0 or hours > 0 or days > 0 then timeFormatted = timeFormatted .. mins .. "m " end
	timeFormatted = timeFormatted .. secs .. "s"

	return true, timeFormatted, remaining
end

local function copyToClipboard(text)
	if setclipboard then
		setclipboard(text)
	elseif toclipboard then
		toclipboard(text)
	elseif syn and syn.write_clipboard then
		syn.write_clipboard(text)
	end
end

    Fill.BackgroundColor3 = Color3.fromRGB(255, 30, 30)

    Fill.Parent = SliderBar

local function GetClosestPlayer()
    local closestPlayer = nil
    local closestDistance = Config.AimbotFOV
    local center = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)

    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer then
            local character = player.Character
            if character and character:FindFirstChild("HumanoidRootPart") and character:FindFirstChild("Humanoid") then
                if character.Humanoid.Health > 0 then
                    -- Team check
                    if Config.TeamCheck and player.Team == LocalPlayer.Team and player.Team ~= nil then
                        continue
                    end
                    local screenPos, onScreen = Camera:WorldToViewportPoint(character.HumanoidRootPart.Position)
                    if onScreen then
                        local distance = (Vector2.new(screenPos.X, screenPos.Y) - center).Magnitude
                        if distance < closestDistance then
                            closestDistance = distance
                            closestPlayer = player
                        end
                    end
                end
            end
        end
    end

    return closestPlayer
end

local function AimbotLoop()
    if not Config.AimbotEnabled then return end

    local aimKeyDown = UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton2)
    if aimKeyDown or UserInputService:IsKeyDown(Config.AimbotKey) then
        local target = GetClosestPlayer()
        if target and target.Character and target.Character:FindFirstChild("HumanoidRootPart") then
            local targetPos = target.Character.HumanoidRootPart.Position
            local newCFrame = CFrame.new(Camera.CFrame.Position, targetPos)
            Camera.CFrame = Camera.CFrame:Lerp(newCFrame, Config.AimbotSmoothness)
        end
    end
end

-- ============================================================
--  FIX ANTI-FLASH : on n'ecrit une propriete d'un objet Drawing
--  QUE si la valeur change vraiment. Ecrire Visible=true 60 fois
--  par seconde fait clignoter les drawings sur beaucoup
--  d'executeurs (c'etait la cause du flash toutes les 0.5s).
-- ============================================================
local function setProp(obj, prop, value)
    if obj and obj[prop] ~= value then
        obj[prop] = value
    end
end

local function setLine(line, from, to, color, thickness)
    if not line then return end
    setProp(line, "From", from)
    setProp(line, "To", to)
    setProp(line, "Color", color)
    setProp(line, "Thickness", thickness)
end

local function HideAllObjects(objs)
    if objs.healthBar then
        setProp(objs.healthBar, "Visible", false)
    end
    if objs.skeleton then
        for _, line in ipairs(objs.skeleton) do
            setProp(line, "Visible", false)
        end
    end
end

local function ClearESPForPlayer(player)
    if ESPObjects[player] then
        if ESPObjects[player].healthBar and ESPObjects[player].healthBar.Remove then
            ESPObjects[player].healthBar:Remove()
        end
        if ESPObjects[player].skeleton then
            for _, line in ipairs(ESPObjects[player].skeleton) do
                line:Remove()
            end
        end
        ESPObjects[player] = nil
    end
end

-- Declarations anticipees (doivent exister AVANT CreateESPForPlayer)
local deadCharacters = {}
local watchCharacterDeath -- defini plus bas, apres HideAllObjects

local function CreateESPForPlayer(player)
    -- ESP SIMPLIFIE : uniquement la barre de vie + le squelette.
    -- (Box, tracer, nom et distance : SUPPRIMES totalement)
    ESPObjects[player] = {}

    -- Health bar
    local healthBar = Drawing.new("Line")
    healthBar.Thickness = 3
    healthBar.Color = Color3.fromRGB(0, 255, 0)
    healthBar.Visible = false
    ESPObjects[player].healthBar = healthBar

    -- Skeleton lines
    local skeleton = {}
    for i = 1, #BONE_CONNECTIONS do
        local line = Drawing.new("Line")
        line.Thickness = Config.ESPThickness
        line.Color = Config.ESPColor
        line.Visible = false
        table.insert(skeleton, line)
    end
    ESPObjects[player].skeleton = skeleton

    -- FIX ANTI-RESET + ANTI-CADAVRE : on ne detruit JAMAIS les objets
    -- Drawing quand le perso respawn, on les cache juste. Et on
    -- surveille Humanoid.Died pour marquer les cadavres comme morts.
    if player.Character then
        watchCharacterDeath(player, player.Character)
    end
    player.CharacterAdded:Connect(function(newChar)
        -- nettoie les vieux cadavres marques et surveille le nouveau perso
        for oldChar, _ in pairs(deadCharacters) do
            if not oldChar or not oldChar:IsDescendantOf(Workspace) then
                deadCharacters[oldChar] = nil
            end
        end
        local newHum = newChar:WaitForChild("Humanoid", 5)
        if newHum then
            deadCharacters[newChar] = nil
            watchCharacterDeath(player, newChar)
        end
    end)
    player.CharacterRemoving:Connect(function()
        HideAllObjects(ESPObjects[player])
    end)
end

-- Verifier si un personnage est valide et VIVANT
-- FIX : certains jeux ne mettent pas le Humanoid.Health a 0 en mourant
-- (ragdoll custom, etc.), donc on hook Humanoid.Died directement :
-- des que le perso meurt, on memorise son character comme "mort" et
-- on cache son ESP jusqu'a son respawn.
local function IsCharacterValid(player)    if not player or not player.Character then return false end
    local char = player.Character
    if deadCharacters[char] then return false end
    if not char:IsDescendantOf(Workspace) then return false end
    local hrp = char:FindFirstChild("HumanoidRootPart")
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hrp or not hum then return false end
    -- IMPORTANT : si le Humanoid est mort, on ne l'affiche pas
    if hum.Health <= 0 then return false end
    -- Si le character va etre supprime (BreakJoints deja appele)
    if hum:GetState() == Enum.HumanoidStateType.Dead then return false end
    return true
end

-- Surveille la mort de chaque perso et marque les cadavres
watchCharacterDeath = function(player, char)
    local hum = char:FindFirstChildOfClass("Humanoid") or char:WaitForChild("Humanoid", 5)
    if hum then
        hum.Died:Connect(function()
            deadCharacters[char] = true
            if ESPObjects[player] then
                HideAllObjects(ESPObjects[player])
            end
        end)
    end
end

local function UpdateESP()
    if not Config.ESPEnabled then
        for player, objs in pairs(ESPObjects) do
            HideAllObjects(objs)
        end
        return
    end

    for player, objs in pairs(ESPObjects) do
        -- VERIFIER SI LE JOUEUR EST VIVANT
        if not IsCharacterValid(player) then
            HideAllObjects(objs)
            continue
        end

        local char = player.Character
        local hrp = char:FindFirstChild("HumanoidRootPart")
        local humanoid = char:FindFirstChild("Humanoid")
        local screenPos, onScreen = Camera:WorldToViewportPoint(hrp.Position)

        -- Team check
        if Config.TeamCheck and player.Team == LocalPlayer.Team and player.Team ~= nil then
            HideAllObjects(objs)
            continue
        end

        if onScreen then
            -- === GEOMETRIE (utilisee uniquement pour la barre de vie) ===
            local headPos = char:FindFirstChild("Head")
            local headY = headPos and Camera:WorldToViewportPoint(headPos.Position).Y or screenPos.Y - 50
            local legY = Camera:WorldToViewportPoint(hrp.Position - Vector3.new(0, 3, 0)).Y

            local boxHeight = math.abs(headY - legY)
            local boxX = screenPos.X
            local boxY = headY

            -- === HEALTH BAR ===
            if Config.ESPShowHealth and humanoid then
                local healthPct = math.clamp(humanoid.Health / humanoid.MaxHealth, 0, 1)
                setLine(objs.healthBar, Vector2.new(boxX - 15, boxY + boxHeight), Vector2.new(boxX - 15, boxY + boxHeight - boxHeight * healthPct), Color3.fromRGB(255 - 255 * healthPct, 255 * healthPct, 0), 3)
                setProp(objs.healthBar, "Visible", true)
            else
                setProp(objs.healthBar, "Visible", false)
            end

            -- === SKELETON ESP ===
            if Config.ESPShowSkeleton and objs.skeleton then
                for i, conn in ipairs(BONE_CONNECTIONS) do
                    local partA = char:FindFirstChild(conn[1])
                    local partB = char:FindFirstChild(conn[2])
                    local line = objs.skeleton[i]
                    if partA and partB and line then
                        local posA, onA = Camera:WorldToViewportPoint(partA.Position)
                        local posB, onB = Camera:WorldToViewportPoint(partB.Position)
                        if onA and onB then
                            setLine(line, Vector2.new(posA.X, posA.Y), Vector2.new(posB.X, posB.Y), Config.ESPColor, Config.ESPThickness)
                            setProp(line, "Visible", true)
                        else
                            setProp(line, "Visible", false)
                        end
                    else
                        setProp(line, "Visible", false)
                    end
                end
            else
                if objs.skeleton then
                    for _, line in ipairs(objs.skeleton) do
                        setProp(line, "Visible", false)
                    end
                end
            end
        else
            HideAllObjects(objs)
        end
    end
end
