        if state then

                BackgroundColor3 = Color3.fromRGB(255, 30, 30)
            }):Play()
            TweenService:Create(ToggleCircle, TweenInfo.new(0.15), {
                Position = UDim2.new(1, -20, 0.5, -9)

        state = not state
        UpdateToggle()
        callback(state)
    end)

local function CreateSlider(text, minVal, maxVal, default, callback)
    local SliderFrame = Instance.new("Frame")
    SliderFrame.Size = UDim2.new(1, 0, 0, 50)
    SliderFrame.BackgroundColor3 = Color3.fromRGB(30, 30, 38)
    SliderFrame.BorderSizePixel = 0
    SliderFrame.LayoutOrder = nextOrder()

    local Corner = Instance.new("UICorner")
    Corner.CornerRadius = UDim.new(0, 6)
    Corner.Parent = SliderFrame

    local Label = Instance.new("TextLabel")
    Label.Size = UDim2.new(0.6, 0, 0, 20)
    Label.Position = UDim2.new(0, 10, 0, 5)
    Label.BackgroundTransparency = 1
    Label.Text = text
    Label.TextColor3 = Color3.fromRGB(255, 255, 255)
    Label.Font = Enum.Font.Gotham
    Label.TextSize = 13
    Label.TextXAlignment = Enum.TextXAlignment.Left
    Label.Parent = SliderFrame

    local ValueLabel = Instance.new("TextLabel")
    ValueLabel.Size = UDim2.new(0, 60, 0, 20)
    ValueLabel.Position = UDim2.new(1, -65, 0, 5)
    ValueLabel.BackgroundTransparency = 1
    ValueLabel.Text = tostring(default)
    ValueLabel.TextColor3 = Color3.fromRGB(255, 60, 60)
    ValueLabel.Font = Enum.Font.GothamBold
    ValueLabel.TextSize = 13
    ValueLabel.TextXAlignment = Enum.TextXAlignment.Right
    ValueLabel.Parent = SliderFrame

    local SliderBar = Instance.new("TextButton")
    SliderBar.Size = UDim2.new(1, -20, 0, 6)
    SliderBar.Position = UDim2.new(0, 10, 0, 33)
    SliderBar.BackgroundColor3 = Color3.fromRGB(50, 50, 55)
    SliderBar.BorderSizePixel = 0
    SliderBar.Text = ""
    SliderBar.AutoButtonColor = false
    SliderBar.Parent = SliderFrame

    local BarCorner = Instance.new("UICorner")
    BarCorner.CornerRadius = UDim.new(1, 0)
    BarCorner.Parent = SliderBar

    local Fill = Instance.new("Frame")
    Fill.Size = UDim2.new((default - minVal) / (maxVal - minVal), 0, 1, 0)
    Fill.BackgroundColor3 = Color3.fromRGB(255, 30, 30)
    Fill.BorderSizePixel = 0
    Fill.Parent = SliderBar

    local FillCorner = Instance.new("UICorner")
    FillCorner.CornerRadius = UDim.new(1, 0)
    FillCorner.Parent = Fill

    local sliderDragging = false

    local function UpdateValue(inputX)
        local rel = math.clamp((inputX - SliderBar.AbsolutePosition.X) / SliderBar.AbsoluteSize.X, 0, 1)
        local value = math.floor(minVal + (maxVal - minVal) * rel + 0.5)
        Fill.Size = UDim2.new(rel, 0, 1, 0)
        ValueLabel.Text = tostring(value)
        callback(value)
    end

    SliderBar.MouseButton1Down:Connect(function()
        sliderDragging = true
        UpdateValue(Mouse.X)
    end)
    UserInputService.InputChanged:Connect(function(input)
        if sliderDragging and input.UserInputType == Enum.UserInputType.MouseMovement then
            UpdateValue(input.Position.X)
        end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            sliderDragging = false
        end
    end)

    SliderFrame.Parent = ScrollingFrame
    return SliderFrame
end

-- ============================================================
--  AIMBOT
-- ============================================================

local function GetClosestPlayer()
    local closestPlayer = nil
    local closestDistance = Config.AimbotFOV
    local center = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)

    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer then
            local character = player.Character
            if character and character:FindFirstChild("HumanoidRootPart") and character:FindFirstChild("Humanoid") then
                if character.Humanoid.Health > 0 then
                    -- Team check
                    if Config.TeamCheck and player.Team == LocalPlayer.Team and player.Team ~= nil then
                        continue
                    end
                    local screenPos, onScreen = Camera:WorldToViewportPoint(character.HumanoidRootPart.Position)
                    if onScreen then
                        local distance = (Vector2.new(screenPos.X, screenPos.Y) - center).Magnitude
                        if distance < closestDistance then
                            closestDistance = distance
                            closestPlayer = player
                        end
                    end
                end
            end
        end
    end

    return closestPlayer
end

local function AimbotLoop()
    if not Config.AimbotEnabled then return end

    local aimKeyDown = UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton2)
    if aimKeyDown or UserInputService:IsKeyDown(Config.AimbotKey) then
        local target = GetClosestPlayer()
        if target and target.Character and target.Character:FindFirstChild("HumanoidRootPart") then
            local targetPos = target.Character.HumanoidRootPart.Position
            local newCFrame = CFrame.new(Camera.CFrame.Position, targetPos)
            Camera.CFrame = Camera.CFrame:Lerp(newCFrame, Config.AimbotSmoothness)
        end
    end
end

local function HideAllObjects(objs)
    for _, obj in ipairs(objs) do
        if obj and obj.Visible ~= nil and typeof(obj) ~= "table" then
            obj.Visible = false
        end
    end
    if objs.skeleton then
        for _, line in ipairs(objs.skeleton) do
            line.Visible = false
        end
    end
end

local function ClearESPForPlayer(player)
    if ESPObjects[player] then
        for _, obj in ipairs(ESPObjects[player]) do
            if obj and obj.Remove then
                obj:Remove()
            end
        end
        if ESPObjects[player].skeleton then
            for _, line in ipairs(ESPObjects[player].skeleton) do
                line:Remove()
            end
        end
        ESPObjects[player] = nil
    end
end

local function CreateESPForPlayer(player)
    ESPObjects[player] = {}

    -- Box ESP (Quad)
    local box = Drawing.new("Quad")
    box.Thickness = Config.ESPThickness
    box.Color = Config.ESPColor
    box.Filled = false
    box.Visible = false
    table.insert(ESPObjects[player], box)

    -- Tracer
    local tracer = Drawing.new("Line")
    tracer.Thickness = Config.ESPThickness
    tracer.Color = Config.ESPColor
    tracer.Visible = false
    table.insert(ESPObjects[player], tracer)

    -- Name
    local nameText = Drawing.new("Text")
    nameText.Size = 14
    nameText.Center = true
    nameText.Outline = true
    nameText.Color = Config.ESPColor
    nameText.Visible = false
    table.insert(ESPObjects[player], nameText)

    -- Distance
    local distText = Drawing.new("Text")
    distText.Size = 12
    distText.Center = true
    distText.Outline = true
    distText.Color = Color3.fromRGB(255, 255, 255)
    distText.Visible = false
    table.insert(ESPObjects[player], distText)

    -- Health bar
    local healthBar = Drawing.new("Line")
    healthBar.Thickness = 3
    healthBar.Color = Color3.fromRGB(0, 255, 0)
    healthBar.Visible = false
    table.insert(ESPObjects[player], healthBar)

    -- Skeleton lines
    local skeleton = {}
    for i = 1, #BONE_CONNECTIONS do
        local line = Drawing.new("Line")
        line.Thickness = Config.ESPThickness
        line.Color = Config.ESPColor
        line.Visible = false
        table.insert(skeleton, line)
    end
    ESPObjects[player].skeleton = skeleton

    player.CharacterAdded:Connect(function()
        ClearESPForPlayer(player)
        CreateESPForPlayer(player)
    end)
    player.CharacterRemoving:Connect(function()
        ClearESPForPlayer(player)
    end)
end

-- Verifier si un personnage est valide et VIVANT
local function IsCharacterValid(player)
    if not player or not player.Character then return false end
    local char = player.Character
    local hrp = char:FindFirstChild("HumanoidRootPart")
    local hum = char:FindFirstChild("Humanoid")
    if not hrp or not hum then return false end
    -- IMPORTANT : si le Humanoid est mort, on ne l'affiche pas
    if hum.Health <= 0 then return false end
    -- Si le character va etre supprime (BreakJoints deja appele)
    if hum:GetState() == Enum.HumanoidStateType.Dead then return false end
    return true
end

local function UpdateESP()
    if not Config.ESPEnabled then
        for player, objs in pairs(ESPObjects) do
            HideAllObjects(objs)
        end
        return
    end

    for player, objs in pairs(ESPObjects) do
        -- VERIFIER SI LE JOUEUR EST VIVANT
        if not IsCharacterValid(player) then
            HideAllObjects(objs)
            continue
        end

        local char = player.Character
        local hrp = char:FindFirstChild("HumanoidRootPart")
        local humanoid = char:FindFirstChild("Humanoid")
        local screenPos, onScreen = Camera:WorldToViewportPoint(hrp.Position)

        -- Team check
        if Config.TeamCheck and player.Team == LocalPlayer.Team and player.Team ~= nil then
            HideAllObjects(objs)
            continue
        end

        if onScreen then
            -- === BOX ESP ===
            local headPos = char:FindFirstChild("Head")
            local headY = headPos and Camera:WorldToViewportPoint(headPos.Position).Y or screenPos.Y - 50
            local legY = Camera:WorldToViewportPoint(hrp.Position - Vector3.new(0, 3, 0)).Y

            local boxHeight = math.abs(headY - legY)
            local boxWidth = boxHeight * 0.5
            local boxX = screenPos.X
            local boxY = headY

            objs[1].PointA = Vector2.new(boxX - boxWidth / 2, boxY)
            objs[1].PointB = Vector2.new(boxX + boxWidth / 2, boxY)
            objs[1].PointC = Vector2.new(boxX + boxWidth / 2, boxY + boxHeight)
            objs[1].PointD = Vector2.new(boxX - boxWidth / 2, boxY + boxHeight)
            objs[1].Visible = true
            objs[1].Color = Config.ESPColor
            objs[1].Thickness = Config.ESPThickness

            -- === TRACER ===
            objs[2].From = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y)
            objs[2].To = Vector2.new(boxX, boxY + boxHeight)
            objs[2].Visible = true
            objs[2].Color = Config.ESPColor
            objs[2].Thickness = Config.ESPThickness

            -- === NAME ===
            if Config.ESPShowName then
                objs[3].Text = player.Name
                objs[3].Position = Vector2.new(boxX, boxY - 15)
                objs[3].Color = Config.ESPColor
                objs[3].Visible = true
            else
                objs[3].Visible = false
            end

            -- === DISTANCE ===
            if Config.ESPShowDistance then
                local dist = math.floor((Camera.CFrame.Position - hrp.Position).Magnitude)
                objs[4].Text = dist .. "m"
                objs[4].Position = Vector2.new(boxX, boxY + boxHeight + 5)
                objs[4].Visible = true
            else
                objs[4].Visible = false
            end

            -- === HEALTH BAR ===
            if Config.ESPShowHealth and humanoid then
                local healthPct = math.clamp(humanoid.Health / humanoid.MaxHealth, 0, 1)
                objs[5].From = Vector2.new(boxX - boxWidth / 2 - 5, boxY + boxHeight)
                objs[5].To = Vector2.new(boxX - boxWidth / 2 - 5, boxY + boxHeight - boxHeight * healthPct)
                objs[5].Color = Color3.fromRGB(255 - 255 * healthPct, 255 * healthPct, 0)
                objs[5].Visible = true
            else
                objs[5].Visible = false
            end

            -- === SKELETON ESP ===
            if Config.ESPShowSkeleton and objs.skeleton then
                for i, conn in ipairs(BONE_CONNECTIONS) do
                    local partA = char:FindFirstChild(conn[1])
                    local partB = char:FindFirstChild(conn[2])
                    local line = objs.skeleton[i]
                    if partA and partB and line then
                        local posA, onA = Camera:WorldToViewportPoint(partA.Position)
                        local posB, onB = Camera:WorldToViewportPoint(partB.Position)
                        if onA and onB then
                            line.From = Vector2.new(posA.X, posA.Y)
                            line.To = Vector2.new(posB.X, posB.Y)
                            line.Color = Config.ESPColor
                            line.Thickness = Config.ESPThickness
                            line.Visible = true
                        else
                            line.Visible = false
                        end
                    else
                        if line then line.Visible = false end
                    end
                end
            else
                if objs.skeleton then
                    for _, line in ipairs(objs.skeleton) do
                        line.Visible = false
                    end
                end
            end
        else
            HideAllObjects(objs)
        end
    end
end
